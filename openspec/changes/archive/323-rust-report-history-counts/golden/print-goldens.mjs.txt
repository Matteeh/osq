// Prints TypeScript's `osq report` goldens for rust-report "Report summary
// equals TypeScript's", through the History counts.
// Run from the repository root after `pnpm build`:
//   node print-goldens.mjs <fixture folder> <golden folder>
// <fixture folder> holds the crate's `tests/fixtures/summary/` projects plus
// `history-project`. Each case runs the built CLI on a fresh copy of its
// fixture project, with the copy as the working directory, a fresh empty HOME
// and no osq config. git ignores `.run/running/`, so the script writes a
// case's lock files into its copy. `<case>.txt` holds `osq report`'s text from
// its first line through the `    Requeued for agent:` line; `<case>.json`
// holds the `completionRate`, `now` and `specs` members of `osq report --json`
// and its `history` member cut to the keys in HISTORY_KEYS, written as
// `serializeSortedJson` writes them. Neither file ends with a newline.
import { spawnSync } from 'node:child_process';
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';

const BIN = path.resolve('dist', 'cli', 'bin.js');
const [fixture, golden] = process.argv.slice(2);
const HISTORY_KEYS = [
  'attempts',
  'cost',
  'deadByReason',
  'preSpawnVerify',
  'rejections',
  'retries',
  'scopeRegressions',
  'unexplainedReruns',
  'verifyRuns',
];

const cases = [
  { name: 'empty', project: null },
  { name: 'report-project', project: 'report-project' },
  {
    name: 'report-project-running',
    project: 'report-project',
    locks: [
      ['003-running', '7', 'not json\n'],
      ['003-running', '8', '{"pid":1,"startedAt":1}\n'],
    ],
  },
  { name: 'all-done', project: 'all-done' },
  { name: 'thirds', project: 'thirds' },
  { name: 'history-project', project: 'history-project' },
];

function run(root, home, args) {
  const result = spawnSync(process.execPath, [BIN, ...args], {
    cwd: root,
    encoding: 'utf8',
    env: { ...process.env, HOME: home, NO_COLOR: '1' },
  });
  if (result.status !== 0) throw new Error(`${args.join(' ')} failed: ${result.stderr}`);
  return result.stdout;
}

function historyText(stdout) {
  const lines = stdout.split('\n');
  const last = lines.findIndex((line) => line.startsWith('    Requeued for agent: '));
  if (last < 0) throw new Error('no requeued line');
  return lines.slice(0, last + 1).join('\n');
}

function sorted(value) {
  if (Array.isArray(value)) return value.map(sorted);
  if (value === null || typeof value !== 'object') return value;
  return Object.fromEntries(
    Object.keys(value)
      .sort()
      .map((key) => [key, sorted(value[key])]),
  );
}

function historyJson(stdout) {
  const report = JSON.parse(stdout);
  const history = {};
  for (const key of HISTORY_KEYS) history[key] = report.history[key];
  const picked = {
    completionRate: report.completionRate,
    history,
    now: report.now,
    specs: report.specs,
  };
  return JSON.stringify(sorted(picked), null, 2);
}

for (const c of cases) {
  const root = fs.mkdtempSync(path.join(os.tmpdir(), 'osq-report-'));
  const home = fs.mkdtempSync(path.join(os.tmpdir(), 'osq-report-home-'));
  if (c.project !== null) fs.cpSync(path.join(fixture, c.project), root, { recursive: true });
  for (const [folder, task, text] of c.locks ?? []) {
    const running = path.join(root, 'openspec', 'changes', folder, '.run', 'running');
    fs.mkdirSync(running, { recursive: true });
    fs.writeFileSync(path.join(running, `${task}.pid`), text);
  }
  fs.writeFileSync(path.join(golden, `${c.name}.txt`), historyText(run(root, home, ['report'])));
  fs.writeFileSync(
    path.join(golden, `${c.name}.json`),
    historyJson(run(root, home, ['report', '--json'])),
  );
  fs.rmSync(root, { recursive: true, force: true });
  fs.rmSync(home, { recursive: true, force: true });
}
