- [ ] `WebServerOptions` gains `site?: WebServerSite`. With a site, `startWebServer` implements web-inspection "Project paths on a server": it strips the `/p/<project>` prefix and hands the rest to today's routing, so the static files, every document, the event stream, and the actions guard, token and method rules work under the prefix; it answers the two redirects with 302 and `Location: /p/<project>/`, serves `GET /p/<project>/api/server` as JSON with `Cache-Control: no-store` through the same JSON helper as the other documents, answers every other path with 404 `{"error":"not found"}`, and returns `url` `http://127.0.0.1:<port>/p/<project>/`.

import type { ServiceRecord, WatcherRecord } from '../run/watch-state.js';

}
/** The `api/server` document. */
export interface WebServerStatus {
  readonly name: string;
  readonly project: string;
  /** `/p/<project>/`. */
  readonly path: string;
  readonly service: ServiceRecord | null;
  readonly watcher: WatcherRecord | null;
  /** Absolute path of watch.log. */
  readonly log: string;
}
export async function readServerStatus(
  projectRoot: string,
  site: WebServerSite,
  home?: string,
): Promise<WebServerStatus>;
