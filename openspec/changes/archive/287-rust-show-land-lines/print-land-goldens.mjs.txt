// Prints TypeScript's `osq show 10` output for each row of rust-show
// "Show land lines equal TypeScript's".
// Run from the repository root after `pnpm build`:
//   node print-land-goldens.mjs <fixture show-project> <golden folder>
// Each case copies the fixture to a fresh scratch root, writes 003-running's
// `.run/running/1.pid` marker, and builds the case's git state with the git
// program: `git init -b main` in the root, commits made with author and
// committer `osq <osq@example.com>`, and for a worktree case a branch
// `osq/010-archived` checked out in the worktree `<root>-wt`. A case that sets
// `vcs` writes `osq.config.ts` with `vcs: { enabled: true, author: "osq <osq@example.com>" }`;
// the Rust test writes `osq.toml` with the same two keys under `[vcs]` instead. HOME is a fresh
// empty folder, so no stacked approval exists. The built CLI runs with the root
// as its working directory, and its stdout is written to <name>.txt.
import { spawnSync } from 'node:child_process';
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';

const BIN = path.resolve('dist', 'cli', 'bin.js');
const [fixture, golden] = process.argv.slice(2);
const ARCHIVED = path.join('openspec', 'changes', 'archive', '010-archived');
const GIT_ENV = {
  GIT_AUTHOR_NAME: 'osq',
  GIT_AUTHOR_EMAIL: 'osq@example.com',
  GIT_COMMITTER_NAME: 'osq',
  GIT_COMMITTER_EMAIL: 'osq@example.com',
};

function git(cwd, ...args) {
  const run = spawnSync('git', args, { cwd, encoding: 'utf8', env: { ...process.env, ...GIT_ENV } });
  if (run.status !== 0) throw new Error(`git ${args.join(' ')}: ${run.stderr}`);
}

function append(file, text) {
  fs.mkdirSync(path.dirname(file), { recursive: true });
  fs.appendFileSync(file, text);
}

// The archived change committed on main: landed.
function landed(root) {
  git(root, 'init', '-q', '-b', 'main');
  git(root, 'add', '-A');
  git(root, 'commit', '-q', '-m', 'all');
}

// main without the archived change and no osq/010-archived branch.
function noBranch(root) {
  const keep = `${root}-keep`;
  fs.renameSync(path.join(root, ARCHIVED), keep);
  landed(root);
  fs.renameSync(keep, path.join(root, ARCHIVED));
}

// main without the archived change; the branch adds it and one line of src/alpha.ts,
// then main gains `moves` commits that each add one line to src/beta.ts.
function worktree(moves) {
  return (root) => {
    const wt = `${root}-wt`;
    const keep = `${root}-keep`;
    fs.renameSync(path.join(root, ARCHIVED), keep);
    landed(root);
    git(root, 'worktree', 'add', '-q', '-b', 'osq/010-archived', wt);
    fs.cpSync(keep, path.join(wt, ARCHIVED), { recursive: true });
    fs.rmSync(keep, { recursive: true, force: true });
    append(path.join(wt, 'src', 'alpha.ts'), 'export const more = 2;\n');
    git(wt, 'add', '-A');
    git(wt, 'commit', '-q', '-m', 'archive 010');
    for (let n = 1; n <= moves; n++) {
      append(path.join(root, 'src', 'beta.ts'), `export const move${n} = ${n};\n`);
      git(root, 'commit', '-q', '-am', `move ${n}`);
    }
  };
}

// No git; the archived change's stream gains a sync and then a stop.
function syncStopped(root) {
  append(
    path.join(root, ARCHIVED, '.run', 'events', 'change.jsonl'),
    '{"type":"synced","timestamp":"2026-10-06T08:00:00.000Z","data":{"defaultBranch":"main","commits":2}}\n' +
      '{"type":"sync_stopped","timestamp":"2026-10-06T09:00:00.000Z","data":{"reason":"conflict","message":"src/alpha.ts conflicts\\nsecond line"}}\n',
  );
}

const cases = [
  { name: 'land-landed-010', setup: landed, vcs: false },
  { name: 'land-no-branch-010', setup: noBranch, vcs: false },
  { name: 'land-no-branch-vcs-010', setup: noBranch, vcs: true },
  { name: 'land-unmoved-010', setup: worktree(0), vcs: true },
  { name: 'land-one-commit-010', setup: worktree(1), vcs: true },
  { name: 'land-two-commits-010', setup: worktree(2), vcs: true },
  { name: 'land-sync-stopped-010', setup: syncStopped, vcs: false },
];

for (const c of cases) {
  const root = fs.mkdtempSync(path.join(os.tmpdir(), 'osq-show-land-'));
  const home = fs.mkdtempSync(path.join(os.tmpdir(), 'osq-show-home-'));
  fs.cpSync(fixture, root, { recursive: true });
  const running = path.join(root, 'openspec', 'changes', '003-running', '.run', 'running');
  fs.mkdirSync(running, { recursive: true });
  fs.writeFileSync(path.join(running, '1.pid'), '{"pid":4242,"startedAt":1759395600000}\n');
  if (c.vcs) {
    fs.writeFileSync(
      path.join(root, 'osq.config.ts'),
      'export default { vcs: { enabled: true, author: "osq <osq@example.com>" } };\n',
    );
  }
  c.setup(root);
  const run = spawnSync(process.execPath, [BIN, 'show', '10'], {
    cwd: root,
    encoding: 'utf8',
    env: { ...process.env, NO_COLOR: '1', HOME: home },
  });
  if (run.status !== 0 || run.stderr !== '') {
    throw new Error(`${c.name}: exit ${run.status} ${run.stderr}`);
  }
  if (run.stdout.includes(root)) throw new Error(`${c.name}: output names the scratch root`);
  fs.writeFileSync(path.join(golden, `${c.name}.txt`), run.stdout);
  for (const target of [root, `${root}-wt`, home]) {
    fs.rmSync(target, { recursive: true, force: true });
  }
}
